\section{AI Approach}

\begin{frame}{HLA-ProtBERT: Multi-Encoder Pipeline}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{itemize}
        \item Encoders: ProtBERT (768), ESM-2 (1280), ProtT5 (1024), Ankh base (768), Ankh large (1536)
        \item Shared encoder base: caching, locus filtering, SSL toggle, optional HF token
        \item GPU auto-detection; batch encoding to handle full IMGT/HLA catalog
        \item Pipeline script orchestrates downloads, embedding generation, and locus plots
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{tcolorbox}[colback=gray!5,colframe=gray!40]

      \forbioinformatics{Transfer learning from vast protein data}
      \end{tcolorbox}
    \end{column}
  \end{columns}
  
  \keytakeaway{ProtBERT understands protein "language" beyond simple sequence similarity.}
\end{frame}

\section{Technical Appendix}

\begin{frame}{Technical Details}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Model Specifications}
      \begin{itemize}\small
        \item ProtBERT: 420M params, 768-dim
        \item ESM-2 (650M): 1280-dim
        \item ProtT5 XL: 1.3B params, 1024-dim
        \item Ankh Base/Large: 50M/650M, 768/1536-dim
        \item Ensemble-ready; embeddings cached under \texttt{data/embeddings/\*}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Implementation}
      \begin{itemize}\small
        \item \texttt{run\_complete\_pipeline\_all\_encoders.sh} orchestrates download \& encoding
        \item Hugging Face downloads with optional HF token; \texttt{--disable-ssl-verify} flag
        \item GPU auto-detection; batch encoding defaults to 8
        \item Caching + resumable runs for embeddings and locus plots
        \item Locus analysis via \texttt{scripts/run\_locus\_analysis.py} (PCA/t-SNE/UMAP)
      \end{itemize}
    \end{column}
  \end{columns}
  
  \textbf{Dimensionality Reduction Parameters}
  \begin{itemize}\small
    \item \textbf{UMAP:} n\_neighbors=15, min\_dist=0.1
    \item \textbf{t-SNE:} perplexity=30, learning\_rate=200
    \item \textbf{PCA:} n\_components=2
  \end{itemize}
\end{frame}
